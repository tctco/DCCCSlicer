cmake_minimum_required(VERSION 3.16)
project(FdgReferenceTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ITK REQUIRED)
if(ITK_USE_FILE)
    include(${ITK_USE_FILE})
endif()
enable_testing()
add_executable(fdg_reference_statistics ../test_fdg_reference_statistics.cpp)
add_test(NAME fdg_reference_statistics COMMAND fdg_reference_statistics)
add_executable(fdg_reference_image test_image.cpp ../../core/common/ImageOps.cpp)
if(TARGET itk::itk)
    target_link_libraries(fdg_reference_image PRIVATE itk::itk)
else()
    target_link_libraries(fdg_reference_image PRIVATE ${ITK_LIBRARIES})
endif()
add_test(NAME fdg_reference_image COMMAND fdg_reference_image)
